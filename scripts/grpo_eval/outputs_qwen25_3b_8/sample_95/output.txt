   \[

   \[

   \]

   \[
   \text{Number of girls in Girl Scouts} = \frac{2}{3} \times 120 = \frac{2 \times 120}{3} = \frac{240}{3} = 80
   \]

   \[
   \text{Number of girls not in Girl Scouts} = \text{Total number of girls} - \text{Number of girls in Girl Scouts} = 120 - 80 = 40
   \]
